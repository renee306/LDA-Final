
# AR(1) Model with 1 factor 3 items (1 person)

model {

  # Observation model
  for (t in 1:Nt){

    for(j in 1:3){
      y[t, j] ~ dnorm(mu.y[t, j], psi.y[j])
    }

    eta[t] ~ dnorm(mu.eta[t], psi.eta)
  }

  # AR(1) structure for latent factor
  mu.eta[1] <- alpha

  for (t in 2:Nt) {
    eta.center[t - 1] <- eta[t - 1] - alpha
    mu.eta[t] <- alpha + beta * eta.center[t - 1]
  }

  # Measurement model (time-invariant loadings)
  for (t in 1:Nt) {
    mu.y[t, 1] <- eta[t]
    mu.y[t, 2] <- nu.y[1] + lambda.y[1] * eta[t]
    mu.y[t, 3] <- nu.y[2] + lambda.y[2] * eta[t]
  }

  # Residual precisions
  for (j in 1:3) {
    psi.y[j] ~ dgamma(1, 1)
  }

  # Latent factor precision
  psi.eta ~ dgamma(1, 1)

  # Time-series parameters
  alpha ~ dnorm(0, 0.1)
  beta  ~ dunif(-1, 1)

  # Free loadings and intercepts
  for(l in 1:2) {
    lambda.y[l] ~ dnorm(0.5, 1)
    nu.y[l] ~ dnorm(0, 0.01)
  }

  # Variance transformations
  for (j in 1:3) {
    sigma.y[j] <- 1 / psi.y[j]
  }

  sigma.eta <- 1 / psi.eta

  # Standardized factor loadings
  lambda.y.strd[1] <- (
    sqrt(sigma.eta) /
    sqrt(sigma.eta + sigma.y[1])
  )

  lambda.y.strd[2] <- (
    lambda.y[1] * sqrt(sigma.eta) /
    sqrt(lambda.y[1]^2 * sigma.eta + sigma.y[2])
  )

  lambda.y.strd[3] <- (
    lambda.y[2] * sqrt(sigma.eta) /
    sqrt(lambda.y[2]^2 * sigma.eta + sigma.y[3])
  )

}
